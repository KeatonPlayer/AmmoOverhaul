cmake_minimum_required(VERSION 3.20)
project(DiverseCasings VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(ShotQueueTests tests/ShotQueueTests.cpp)
target_include_directories(ShotQueueTests PRIVATE src)
# Keep assert-based behavioral checks enabled in Release too.
if(MSVC)
  target_compile_options(ShotQueueTests PRIVATE /UNDEBUG)
else()
  target_compile_options(ShotQueueTests PRIVATE -UNDEBUG)
endif()
add_test(NAME ShotQueueTests COMMAND ShotQueueTests)
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
    message(FATAL_ERROR "Use -A Win32; Fallout New Vegas is x86.")
  endif()
  add_library(DiverseCasings SHARED src/Plugin.cpp src/exports.def)
  set_property(TARGET DiverseCasings PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(MSVC)
    target_compile_options(DiverseCasings PRIVATE /W4 /permissive- /EHsc)
  endif()
  install(TARGETS DiverseCasings RUNTIME DESTINATION NVSE/Plugins)
  install(FILES package/NVSE/Plugins/DiverseCasings.ini DESTINATION NVSE/Plugins)
  install(FILES README.md DESTINATION .)
endif()
